-- ============================================================
-- 1. OVERALL BUSINESS PERFORMANCE
-- ============================================================

SELECT
    COUNT(*) AS total_transactions,
    ROUND(SUM(revenue), 2) AS total_revenue,
    ROUND(SUM(margin), 2) AS total_margin,
    ROUND(
        SUM(margin) * 100.0 /
        NULLIF(SUM(revenue), 0),
        2
    ) AS margin_pct
FROM public.staging_fmcg;


-- ============================================================
-- 2. HIGHEST-REVENUE CITY
-- ============================================================

SELECT
    city,
    ROUND(SUM(revenue), 2) AS revenue,
    ROUND(SUM(margin), 2) AS margin
FROM public.staging_fmcg
GROUP BY city
ORDER BY revenue DESC
LIMIT 1;


-- ============================================================
-- 3. HIGHEST-REVENUE CATEGORY
-- ============================================================

SELECT
    category,
    ROUND(SUM(revenue), 2) AS revenue,
    ROUND(SUM(margin), 2) AS margin
FROM public.staging_fmcg
GROUP BY category
ORDER BY revenue DESC
LIMIT 1;


-- ============================================================
-- 4. HIGHEST-MARGIN CITY
-- ============================================================

SELECT
    city,
    ROUND(
        SUM(margin) * 100.0 /
        NULLIF(SUM(revenue), 0),
        2
    ) AS margin_pct
FROM public.staging_fmcg
GROUP BY city
ORDER BY margin_pct DESC
LIMIT 1;


-- ============================================================
-- 5. HIGHEST-MARGIN CATEGORY
-- ============================================================

SELECT
    category,
    ROUND(
        SUM(margin) * 100.0 /
        NULLIF(SUM(revenue), 0),
        2
    ) AS margin_pct
FROM public.staging_fmcg
GROUP BY category
ORDER BY margin_pct DESC
LIMIT 1;


-- ============================================================
-- 6. TOP CITY × CATEGORY BY REVENUE
-- ============================================================

SELECT
    city,
    category,
    ROUND(SUM(revenue), 2) AS revenue,
    ROUND(SUM(margin), 2) AS margin
FROM public.staging_fmcg
GROUP BY
    city,
    category
ORDER BY revenue DESC
LIMIT 1;


-- ============================================================
-- 7. TOP CITY × CATEGORY BY ABSOLUTE MARGIN
-- ============================================================

SELECT
    city,
    category,
    ROUND(SUM(revenue), 2) AS revenue,
    ROUND(SUM(margin), 2) AS margin
FROM public.staging_fmcg
GROUP BY
    city,
    category
ORDER BY margin DESC
LIMIT 1;
