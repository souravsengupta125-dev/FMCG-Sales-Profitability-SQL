-- ============================================================
-- 1. RANK CITIES BY REVENUE
-- Business Question:
-- How do cities rank based on total revenue?
-- ============================================================

SELECT
    city,
    ROUND(revenue, 2) AS revenue,
    RANK() OVER (
        ORDER BY revenue DESC
    ) AS revenue_rank
FROM (
    SELECT
        city,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY city
) AS city_revenue
ORDER BY revenue_rank;


-- ============================================================
-- 2. RANK CATEGORIES BY REVENUE
-- Business Question:
-- How do FMCG categories rank based on revenue?
-- ============================================================

SELECT
    category,
    ROUND(revenue, 2) AS revenue,
    RANK() OVER (
        ORDER BY revenue DESC
    ) AS revenue_rank
FROM (
    SELECT
        category,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY category
) AS category_revenue
ORDER BY revenue_rank;


-- ============================================================
-- 3. RANK CATEGORIES WITHIN EACH CITY
-- Business Question:
-- What is the revenue ranking of categories within each city?
-- ============================================================

SELECT
    city,
    category,
    ROUND(revenue, 2) AS revenue,
    RANK() OVER (
        PARTITION BY city
        ORDER BY revenue DESC
    ) AS category_rank
FROM (
    SELECT
        city,
        category,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY
        city,
        category
) AS city_category
ORDER BY
    city,
    category_rank;


-- ============================================================
-- 4. TOP CATEGORY IN EACH CITY
-- Business Question:
-- Which category generates the highest revenue in each city?
-- ============================================================

SELECT
    city,
    category,
    ROUND(revenue, 2) AS revenue
FROM (
    SELECT
        city,
        category,
        SUM(revenue) AS revenue,
        ROW_NUMBER() OVER (
            PARTITION BY city
            ORDER BY SUM(revenue) DESC
        ) AS category_rank
    FROM public.staging_fmcg
    GROUP BY
        city,
        category
) AS ranked_categories
WHERE category_rank = 1
ORDER BY revenue DESC;


-- ============================================================
-- 5. RANK CITIES WITHIN EACH CATEGORY
-- Business Question:
-- How do cities rank within each FMCG category?
-- ============================================================

SELECT
    category,
    city,
    ROUND(revenue, 2) AS revenue,
    RANK() OVER (
        PARTITION BY category
        ORDER BY revenue DESC
    ) AS city_rank
FROM (
    SELECT
        category,
        city,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY
        category,
        city
) AS category_city
ORDER BY
    category,
    city_rank;


-- ============================================================
-- 6. TOP CITY FOR EACH CATEGORY
-- Business Question:
-- Which city is the strongest market for each category?
-- ============================================================

SELECT
    category,
    city,
    ROUND(revenue, 2) AS revenue
FROM (
    SELECT
        category,
        city,
        SUM(revenue) AS revenue,
        ROW_NUMBER() OVER (
            PARTITION BY category
            ORDER BY SUM(revenue) DESC
        ) AS city_rank
    FROM public.staging_fmcg
    GROUP BY
        category,
        city
) AS ranked_cities
WHERE city_rank = 1
ORDER BY revenue DESC;


-- ============================================================
-- 7. REVENUE CONTRIBUTION BY CITY
-- Business Question:
-- What percentage of total company revenue comes from each city?
-- ============================================================

SELECT
    city,
    ROUND(revenue, 2) AS revenue,
    ROUND(
        revenue * 100.0 /
        SUM(revenue) OVER (),
        2
    ) AS revenue_contribution_pct
FROM (
    SELECT
        city,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY city
) AS city_revenue
ORDER BY revenue DESC;


-- ============================================================
-- 8. REVENUE CONTRIBUTION BY CATEGORY
-- Business Question:
-- What percentage of total company revenue comes from each
-- category?
-- ============================================================

SELECT
    category,
    ROUND(revenue, 2) AS revenue,
    ROUND(
        revenue * 100.0 /
        SUM(revenue) OVER (),
        2
    ) AS revenue_contribution_pct
FROM (
    SELECT
        category,
        SUM(revenue) AS revenue
    FROM public.staging_fmcg
    GROUP BY category
) AS category_revenue
ORDER BY revenue DESC;


-- ============================================================
-- 9. TOP 3 CATEGORIES IN EACH CITY
-- Business Question:
-- Which three categories generate the most revenue in each city?
-- ============================================================

SELECT
    city,
    category,
    ROUND(revenue, 2) AS revenue,
    category_rank
FROM (
    SELECT
        city,
        category,
        SUM(revenue) AS revenue,
        ROW_NUMBER() OVER (
            PARTITION BY city
            ORDER BY SUM(revenue) DESC
        ) AS category_rank
    FROM public.staging_fmcg
    GROUP BY
        city,
        category
) AS ranked_categories
WHERE category_rank <= 3
ORDER BY
    city,
    category_rank;


-- ============================================================
-- 10. HIGHEST-MARGIN CITY-CATEGORY SEGMENTS
-- Business Question:
-- Which city-category combinations have the strongest margins?
-- ============================================================

SELECT
    city,
    category,
    ROUND(revenue, 2) AS revenue,
    ROUND(margin, 2) AS margin,
    ROUND(
        margin * 100.0 /
        NULLIF(revenue, 0),
        2
    ) AS margin_pct,
    RANK() OVER (
        ORDER BY
            margin * 100.0 /
            NULLIF(revenue, 0) DESC
    ) AS margin_rank
FROM (
    SELECT
        city,
        category,
        SUM(revenue) AS revenue,
        SUM(margin) AS margin
    FROM public.staging_fmcg
    GROUP BY
        city,
        category
) AS segment_performance
ORDER BY margin_rank;


-- ============================================================
-- 11. RANK CITIES BY MARGIN PERCENTAGE
-- Business Question:
-- Which cities are most efficient in converting revenue into
-- margin?
-- ============================================================

SELECT
    city,
    ROUND(revenue, 2) AS revenue,
    ROUND(margin, 2) AS margin,
    ROUND(
        margin * 100.0 /
        NULLIF(revenue, 0),
        2
    ) AS margin_pct,
    RANK() OVER (
        ORDER BY
            margin * 100.0 /
            NULLIF(revenue, 0) DESC
    ) AS margin_rank
FROM (
    SELECT
        city,
        SUM(revenue) AS revenue,
        SUM(margin) AS margin
    FROM public.staging_fmcg
    GROUP BY city
) AS city_performance
ORDER BY margin_rank;
